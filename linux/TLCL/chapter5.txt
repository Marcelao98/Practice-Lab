  498  def normalize(name):
  499      name = unicodedata.normalize("NFKD", name)
  500      name = "".join(c for c in name if not unicodedata.combining(c))
  501      name = name.lower()
  502      name = re.sub(r"\([^)]*\)|\[[^]]*\]|\{[^}]*\}", " ", name)
  503      name = re.sub(
  504          r"\b(usa|europe|japan|world|rev\s*\d*|"
  505          r"version\s*\d*|u|e|j|en|br|pt-br|good|bad)\b",
  506          " ", name
  507      )
  508      return re.sub(r"[^a-z0-9]", "", name)
  509  def find_matches(game, files):
  510      variants = [game] + ALIASES.get(game, [])
  511      matches = []
  512      for file in files:
  513          filename = normalize(file.stem)
  514          for variant in variants:
  515              target = normalize(variant)
  516              score = SequenceMatcher(None, filename, target).ratio()
  517              if filename == target:
  518                  score = 1.0
  519              elif len(target) >= 7 and target in filename:
  520                  score = max(score, 0.94)
  521              if score >= 0.86:
  522                  matches.append((score, file))
  523                  break
  524      return sorted(matches, key=lambda x: x[0], reverse=True)
  525  def main():
  526      if not ROM_DIR.is_dir():
  527          print(f"ERRO: pasta não encontrada: {ROM_DIR}")
  528          return
  529      REPORT_DIR.mkdir(parents=True, exist_ok=True)
  530      files = [
  531          p for p in ROM_DIR.rglob("*")
  532          if p.is_file() and p.suffix.lower() in EXTENSIONS
  533      ]
  534      print(f"\nROMs encontradas: {len(files)}")
  535      print("Analisando catálogo SNES...\n")
  536      results = []
  537      found_total = 0
  538      missing_total = 0
  539      for category, games in GAMES.items():
  540          print(f"\n===== {category} =====")
  541          for game in games:
  542              matches = find_matches(game, files)
  543              if matches:
  544                  found_total += 1
  545                  status = "ENCONTRADO"
  546                  filename = matches[0][1].name
  547                  print(f"[OK] {game} -> {filename}")
  548                  if len(matches) > 1:
  549                      print(f"     Atenção: {len(matches)} correspondências possíveis")
  550              else:
  551                  missing_total += 1
  552                  status = "FALTANDO"
  553                  filename = ""
  554                  print(f"[FALTA] {game}")
  555              results.append({
  556                  "Categoria": category,
  557                  "Jogo": game,
  558                  "Status": status,
  559                  "Arquivo": filename,
  560                  "Correspondencias": len(matches)
  561              })
  562      total = len(results)
  563      coverage = found_total / total * 100 if total else 0
  564      timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
  565      txt_path = REPORT_DIR / f"SNES_Curadoria_{timestamp}.txt"
  566      csv_path = REPORT_DIR / f"SNES_Curadoria_{timestamp}.csv"
  567      with open(csv_path, "w", newline="", encoding="utf-8-sig") as f:
  568          writer = csv.DictWriter(f, fieldnames=results[0].keys())
  569          writer.writeheader()
  570          writer.writerows(results)
  571      with open(txt_path, "w", encoding="utf-8") as f:
  572          f.write("RELATORIO DE CURADORIA SNES\n")
  573          f.write("=" * 45 + "\n")
  574          f.write(f"Pasta: {ROM_DIR}\n")
  575          f.write(f"Arquivos analisados: {len(files)}\n")
  576          f.write(f"Jogos no catálogo: {total}\n")
  577          f.write(f"Encontrados: {found_total}\n")
  578          f.write(f"Faltantes: {missing_total}\n")
  579          f.write(f"Cobertura: {coverage:.1f}%\n")
  580          for category in GAMES:
  581              f.write(f"\n\n### {category}\n")
  582              for r in results:
  583                  if r["Categoria"] == category:
  584                      f.write(
  585                          f"[{'OK' if r['Status'] == 'ENCONTRADO' else 'FALTA'}] "
  586                          f"{r['Jogo']}"
  587                      )
  588                      if r["Arquivo"]:
  589                          f.write(f" -> {r['Arquivo']}")
  590                      f.write("\n")
